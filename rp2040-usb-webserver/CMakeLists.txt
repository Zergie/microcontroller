cmake_minimum_required(VERSION 3.13)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(PICO_BOARD pico CACHE STRING "Board type")
include(../shared/rp2040-usb-web/pico_sdk_import.cmake)
project(rp2040_usb_webserver C CXX ASM)
pico_sdk_init()
include(../shared/rp2040-usb-web/CMakeLists.txt)
add_executable(rp2040_usb_webserver main.c gpio_control.c gpio_api.c web_routes.c)
target_include_directories(rp2040_usb_webserver PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(rp2040_usb_webserver hardware_pwm hardware_clocks hardware_adc)
target_usb_web(rp2040_usb_webserver "${CMAKE_CURRENT_LIST_DIR}/web")
pico_set_program_name(rp2040_usb_webserver "rp2040_usb_webserver")
pico_set_program_version(rp2040_usb_webserver "0.1")
